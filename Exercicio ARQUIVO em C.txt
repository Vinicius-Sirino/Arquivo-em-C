#include<stdio.h>
#include<stdlib.h>
#include<string.h>

struct pessoa{
    int codigo;

};

struct pessoa funcionario;

    char numstru[20];

    FILE*fptr;
    fptr = fopen("arquivo", "ab");
    if(fptr ==NULL){
        printf("\n erro na abertura do arquivo");
        system("pause");
    }
    else{
        printf("\n informe o codigo...");
        scanf("%d", &funcionario.codigo);
        printf("\n informe o nome: ");
        scanf(" %[^\n]", funcionario.nome);
        printf("\n informe o salario...");
        scanf("%f", &funcionario.salario);
        fwrite(&funcionario, sizeof(funcionario), 1, fptr);
        printf("\n gravado com sucesso...");
        system("pause");
    }

}

void pesquisa(){
    char numstru[20];
    int op, achou;

    if(fptr == NULL){
        printf("\n problema com abertura do arquivo");
        system("pause");
    }
    else{
        achou = 0;
        printf("\n informe o nome a ser pesquisado: ");
        scanf(" %[^\n]", numstru);
        while(fread(&funcionario, sizeof(funcionario), 1, fptr)){
            if(strcmp(funcionario.nome, numstru) == 0){
                printf("\n=== REGISTRO ENCONTRADO ===");
                printf("\nCodigo: %d", funcionario.codigo);
                printf("\nNome: %s", funcionario.nome);
                printf("\nSalario: %.2f", funcionario.salario);

            }
        }
        if(achou == 0){
            printf("\n registro nao esta constando no arquivo");

        }

}
